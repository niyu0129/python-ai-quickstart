# 第 7 天：项目一 —— 命令行聊天机器人

> 今天把第 6 天的 `LLM` 类用成一个**真正能用的产品**：流式输出、多轮记忆、成本统计、历史裁剪、会话保存。

---

## 1. 先想清楚：一个"能用"的聊天机器人需要什么

| 能力 | 为什么需要 | 实现手段 |
|---|---|---|
| 多轮记忆 | 不然它不记得你上句说了什么 | 维护 `messages` 列表 |
| 流式输出 | 不然等 10 秒没反应，像卡死 | `stream=True` |
| 历史裁剪 | 不然越聊越贵，最后超上下文上限 | 只保留最近 N 轮 |
| 成本显示 | 不然不知道烧了多少钱 | 累计 `usage` |
| 命令系统 | 清空、保存、退出 | 解析 `/` 开头的输入 |
| 异常兜底 | 网络断了不能崩 | `try/except` + 重试 |

**这六条就是所有对话产品的骨架**，Web 版、App 版都一样，只是外壳不同。

新建 `chatbot.py`：

```python
# chatbot.py —— 一个功能完整的命令行 AI 助手
import os
import sys
import json
import time
from pathlib import Path
from datetime import datetime

from dotenv import load_dotenv
from openai import OpenAI, APIError, APITimeoutError, RateLimitError

load_dotenv()

# ---------- 配置 ----------
MODEL = os.getenv("OPENAI_MODEL", "gpt-4o-mini")
MAX_TURNS = 10          # 保留最近 10 轮（一轮 = user + assistant）
MAX_TOKENS = 1024

# 每 100 万 token 的价格（美元），按你的厂商改
PRICING = {
    "gpt-4o-mini":   {"in": 0.15, "out": 0.60},
    "deepseek-chat": {"in": 0.27, "out": 1.10},
}
DEFAULT_PRICE = {"in": 0.0, "out": 0.0}     # 本地模型免费

SYSTEM_PROMPT = """你是一个耐心、直接的中文技术助手。
要求：
1. 回答简洁，先给结论再给解释。
2. 涉及代码时给出可运行的完整示例。
3. 不确定的事情明确说"不确定"，不要编造。
"""


class ChatBot:
    def __init__(self, model: str = MODEL, system_prompt: str = SYSTEM_PROMPT,
                 max_turns: int = MAX_TURNS):
        self.client = OpenAI(
            api_key=os.getenv("OPENAI_API_KEY"),
            base_url=os.getenv("OPENAI_BASE_URL"),
            timeout=90.0,
            max_retries=2,
        )
        self.model = model
        self.system_prompt = system_prompt
        self.max_turns = max_turns
        self.messages: list[dict] = []       # 不含 system 的历史
        self.prompt_tokens = 0
        self.completion_tokens = 0
        self.turns = 0

    # ---------- 历史管理 ----------
    def _trim(self) -> None:
        """只保留最近 max_turns 轮，防止上下文无限膨胀。"""
        limit = self.max_turns * 2
        if len(self.messages) > limit:
            removed = len(self.messages) - limit
            self.messages = self.messages[-limit:]
            print(f"\033[90m[已裁剪 {removed} 条较早的消息以控制成本]\033[0m")

    def _payload(self) -> list[dict]:
        """组装发给模型的完整消息（system 永远在最前面）。"""
        return [{"role": "system", "content": self.system_prompt}] + self.messages

    # ---------- 核心：流式对话 ----------
    def ask(self, question: str) -> str:
        self.messages.append({"role": "user", "content": question})
        self._trim()

        print("\033[36mAI:\033[0m ", end="", flush=True)
        t0 = time.time()
        pieces: list[str] = []

        try:
            stream = self.client.chat.completions.create(
                model=self.model,
                messages=self._payload(),
                stream=True,
                temperature=0.7,
                max_tokens=MAX_TOKENS,
                stream_options={"include_usage": True},   # 让最后一帧带 usage
            )
            for chunk in stream:
                if getattr(chunk, "usage", None):         # 用量帧
                    self.prompt_tokens += chunk.usage.prompt_tokens
                    self.completion_tokens += chunk.usage.completion_tokens
                if not chunk.choices:
                    continue
                piece = chunk.choices[0].delta.content
                if piece:
                    pieces.append(piece)
                    print(piece, end="", flush=True)

        except RateLimitError:
            print("\n\033[31m[触发限流，请稍后再试]\033[0m")
            self.messages.pop()                            # 回滚这条 user 消息
            return ""
        except APITimeoutError:
            print("\n\033[31m[请求超时，检查网络或换个模型]\033[0m")
            self.messages.pop()
            return ""
        except APIError as e:
            print(f"\n\033[31m[API 错误] {e}\033[0m")
            self.messages.pop()
            return ""
        except KeyboardInterrupt:
            print("\n\033[90m[已中断生成]\033[0m")

        answer = "".join(pieces)
        elapsed = time.time() - t0

        if answer:
            self.messages.append({"role": "assistant", "content": answer})
            self.turns += 1

        print(f"\n\033[90m({elapsed:.1f}s)\033[0m\n")
        return answer

    # ---------- 统计 ----------
    @property
    def cost(self) -> float:
        p = PRICING.get(self.model, DEFAULT_PRICE)
        return (self.prompt_tokens * p["in"] + self.completion_tokens * p["out"]) / 1_000_000

    def stats(self) -> str:
        total = self.prompt_tokens + self.completion_tokens
        return (f"轮数 {self.turns} | 历史 {len(self.messages)} 条 | "
                f"tokens {total}（入 {self.prompt_tokens} / 出 {self.completion_tokens}）| "
                f"约 ${self.cost:.5f}")

    # ---------- 命令 ----------
    def clear(self) -> None:
        self.messages.clear()
        print("\033[90m[对话历史已清空，system 人设保留]\033[0m")

    def save(self, path: str | None = None) -> Path:
        path = path or f"chat_{datetime.now():%Y%m%d_%H%M%S}.json"
        p = Path(path)
        p.write_text(json.dumps({
            "model": self.model,
            "system": self.system_prompt,
            "messages": self.messages,
            "prompt_tokens": self.prompt_tokens,
            "completion_tokens": self.completion_tokens,
        }, ensure_ascii=False, indent=2), encoding="utf-8")
        print(f"\033[90m[已保存到 {p.resolve()}]\033[0m")
        return p

    def load(self, path: str) -> None:
        data = json.loads(Path(path).read_text(encoding="utf-8"))
        self.messages = data["messages"]
        self.system_prompt = data.get("system", self.system_prompt)
        self.prompt_tokens = data.get("prompt_tokens", 0)
        self.completion_tokens = data.get("completion_tokens", 0)
        print(f"\033[90m[已载入 {len(self.messages)} 条消息]\033[0m")


HELP = """
可用命令：
  /help        显示帮助
  /clear       清空对话历史
  /stats       查看 token 与花费
  /save [文件]  保存对话（默认自动命名）
  /load 文件    载入对话
  /system 文本  修改人设（会清空历史）
  /quit        退出
直接输入内容即可对话。Ctrl+C 可中断本次生成。
"""


def main() -> None:
    if not os.getenv("OPENAI_API_KEY"):
        print("❌ 未检测到 OPENAI_API_KEY，请在项目根目录建 .env 文件并写入密钥。")
        sys.exit(1)

    bot = ChatBot()
    model_name = bot.model
    print("\033[35m" + "=" * 56)
    print(f"  AI 命令行助手  |  模型：{model_name}")
    print("=" * 56 + "\033[0m")
    print("输入 /help 查看命令，/quit 退出\n")

    while True:
        try:
            user_input = input("\033[32m你:\033[0m ").strip()
        except (EOFError, KeyboardInterrupt):
            print("\n再见！")
            break

        if not user_input:
            continue

        # 命令处理
        if user_input.startswith("/"):
            cmd, _, arg = user_input.partition(" ")
            cmd = cmd.lower()
            if cmd in ("/quit", "/exit", "/q"):
                print(f"\n本次会话：{bot.stats()}")
                print("再见！")
                break
            elif cmd == "/help":
                print(HELP)
            elif cmd == "/clear":
                bot.clear()
            elif cmd == "/stats":
                print("\033[90m" + bot.stats() + "\033[0m")
            elif cmd == "/save":
                bot.save(arg.strip() or None)
            elif cmd == "/load":
                if not arg.strip():
                    print("用法：/load 文件名.json")
                elif not Path(arg.strip()).exists():
                    print("文件不存在")
                else:
                    bot.load(arg.strip())
            elif cmd == "/system":
                if arg.strip():
                    bot.system_prompt = arg.strip()
                    bot.messages.clear()
                    print("\033[90m[人设已更新，历史已清空]\033[0m")
                else:
                    print("用法：/system 你是一个...")
            else:
                print(f"未知命令：{cmd}，输入 /help 查看")
            continue

        bot.ask(user_input)


if __name__ == "__main__":
    main()
```


## 3. 关键实现点讲解

### 3.1 为什么 `system` 不放进 `self.messages`？

因为 `/clear` 时要清空对话但**保留人设**。把 system 单独存，每次请求时临时拼在最前面：

```python
def _payload(self):
    return [{"role": "system", "content": self.system_prompt}] + self.messages
```

### 3.2 `messages.pop()` 回滚：出错时不留下脏数据

请求失败了，但 user 消息已经 append 进去了。如果不 pop，下次请求会把这条失败的问题一起发过去，模型会困惑。**这是很多教程会漏掉的细节。**

```python
except APIError:
    self.messages.pop()      # ★ 回滚
    return ""
```

### 3.3 历史裁剪：`_trim()`

```python
limit = self.max_turns * 2          # 一轮包含 user + assistant
if len(self.messages) > limit:
    self.messages = self.messages[-limit:]     # 保留最近 limit 条
```

> ⚠️ **更讲究的做法是"摘要压缩"**：把裁掉的历史用模型总结成一段话，作为新的 system 附加内容。这样既省钱又保留长期记忆。这留给你当挑战题。

### 3.4 token 从哪里来？

我们在 `stream_options` 里开了 `include_usage`，流式请求的**最后一帧**会带 `usage` 对象。所以要**在循环里累加**，而不是循环结束后取一次：

```python
for chunk in stream:
    if getattr(chunk, "usage", None):
        self.prompt_tokens += chunk.usage.prompt_tokens
        self.completion_tokens += chunk.usage.completion_tokens
```

### 3.5 彩色输出的原理

终端里 `\033[36m ... \033[0m` 是 ANSI 转义码：

| 码 | 颜色 | 码 | 效果 |
|---|---|---|---|
| `\033[31m` | 红 | `\033[0m` | 重置 |
| `\033[32m` | 绿 | `\033[1m` | 加粗 |
| `\033[35m` | 紫 | `\033[90m` | 灰色（次要信息） |
| `\033[36m` | 青 | | |

> Windows Terminal 支持良好；老版本 cmd 可能显示乱码，换成 Windows Terminal 即可。

```python
def summarize_and_trim(self):
    if len(self.messages) <= self.max_turns * 2:
        return
    old, self.messages = self.messages[:-self.max_turns * 2], self.messages[-self.max_turns * 2:]
    history_text = "\n".join(f"{m['role']}: {m['content'][:200]}" for m in old)
    summary = self.client.chat.completions.create(
        model=self.model,
        messages=[{"role": "user", "content": f"用 3 句话总结这段对话的要点：\n{history_text}"}],
        temperature=0,
    ).choices[0].message.content
    self.system_prompt = SYSTEM_PROMPT + f"\n\n【此前对话摘要】\n{summary}"
```
